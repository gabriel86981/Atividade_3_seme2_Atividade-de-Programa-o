package calculo_rapido;

import java.util.Scanner;

public class Main {

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        int s, a, b;
        int quantidade = 0;

        s = scanner.nextInt();
        a = scanner.nextInt();
        b = scanner.nextInt();

        while (a <= b) {

            int numero = a;
            int soma = 0;

            while (numero > 0) {
                soma = soma + numero % 10;
                numero = numero / 10;
            }

            if (soma == s) {
                quantidade++;
            }

            a++;
        }

        System.out.println(quantidade);

     
    }
}

package corrida;

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        int n1, d1, v1;
        int n2, d2, v2;

        double tempo1, tempo2;

        n1 = scanner.nextInt();
        d1 = scanner.nextInt();
        v1 = scanner.nextInt();

        n2 = scanner.nextInt();
        d2 = scanner.nextInt();
        v2 = scanner.nextInt();

        tempo1 = (double) d1 / (v1 * 1000);
        tempo2 = (double) d2 / (v2 * 1000);

        if (tempo1 < tempo2) {
            System.out.println(n1);
        } else {
            System.out.println(n2);
        }

           }
